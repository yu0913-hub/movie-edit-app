    async function loadAllData(keepSelection, silent) {
      if (!db) return;
      const prevId = keepSelection ? document.getElementById('projectSelect').value : '';
      const btn = document.getElementById('reloadBtn');
      const ico = document.getElementById('reloadIcon');
      if (!silent) { btn.disabled = true; ico.classList.add('spinning'); }

      try {
        const results = await Promise.all([
          db.from('projects').select('*').order('created_at', { ascending: true }),
          db.from('members').select('*').order('created_at', { ascending: true }),
          db.from('tasks').select('*').order('created_at', { ascending: true }),
          db.from('memos').select('*').order('created_at', { ascending: true }),
          db.from('assets').select('*').order('created_at', { ascending: true }),
          db.from('credits').select('*').order('created_at', { ascending: true })
        ]);

        for (let i = 0; i < results.length; i++) {
          if (results[i].error) throw results[i].error;
        }

        allProjects = results[0].data || [];
        allMembers = results[1].data || [];
        allTasks = results[2].data || [];
        allMemos = results[3].data || [];
        allAssets = results[4].data || [];
        allCredits = results[5].data || [];

        document.getElementById('syncDot').classList.add('online');
        if (document.getElementById('syncText').textContent === '接続中...') {
          document.getElementById('syncText').textContent = '接続OK';
        }

        renderProjectSelect(prevId);
        if (document.getElementById('membersModal').classList.contains('open')) {
          renderMembersListInModal();
        }
        if (keepSelection && !silent) showToast('最新データに更新しました');
      } catch (err) {
        showHomeError('Supabase読込エラー：' + (err.message || err));
      } finally {
        btn.disabled = false;
        ico.classList.remove('spinning');
      }
    }

    // ===== 3階層進捗計算 =====
    function getEffectiveTaskProgress(task) {
      const children = allTasks.filter(function(c) { return c.parent_id === task.id; });
      if (children.length === 0) return Number(task.progress) || 0;
      const sum = children.reduce(function(acc, c) { return acc + (Number(c.progress) || 0); }, 0);
      return Math.round(sum / children.length);
    }

    function calcProjectAvgProgress(projectId) {
      const parents = allTasks.filter(function(t) { return t.project_id === projectId && !t.parent_id; });
      if (parents.length === 0) return 0;
      const sum = parents.reduce(function(acc, p) { return acc + getEffectiveTaskProgress(p); }, 0);
      return Math.round(sum / parents.length);
    }

    function getLeafTasks(projectId) {
      const pTasks = allTasks.filter(function(t) { return t.project_id === projectId; });
      const parentIdsWithChildren = new Set();
      pTasks.forEach(function(t) { if (t.parent_id) parentIdsWithChildren.add(t.parent_id); });
      return pTasks.filter(function(t) { return !parentIdsWithChildren.has(t.id); });
    }

    function formatTaskDisplayName(task) {
      if (!task) return '';
      return (task.label ? '[' + task.label + '] ' : '') + task.name;
    }

    function getFullTaskPath(taskId) {
      const t = allTasks.find(function(x) { return x.id === taskId; });
      if (!t) return '';
      if (t.parent_id) {
        const parent = allTasks.find(function(x) { return x.id === t.parent_id; });
        return (parent ? formatTaskDisplayName(parent) + ' ＞ ' : '') + formatTaskDisplayName(t);
      }
      return formatTaskDisplayName(t);
    }

        const prevPct = task.progress;
        const nextStage = sel.value;

        // 先に画面を即座に更新（体感0秒）
        task.stage = nextStage;
        task.progress = nextPct;
        refreshAllViews();

        const res = await db.from('tasks').update({ stage: nextStage, progress: nextPct }).eq('id', task.id);
        if (res.error) {
          task.stage = prevStage;
          task.progress = prevPct;
          refreshAllViews();
          alert('更新失敗：' + res.error.message);
        } else {
          showToast(task.name + ' を ' + nextPct + '% に更新');
        }
      };

    }

    function parseTimecodeToSec(tc) {
      const s = (tc || '').trim();
      if (!s) return Number.MAX_SAFE_INTEGER;
      const parts = s.split(':').map(Number);
      if (parts.some(isNaN)) return Number.MAX_SAFE_INTEGER - 1;
      if (parts.length === 2) return parts[0] * 60 + parts[1];
      if (parts.length === 3) return parts[0] * 3600 + parts[1] * 60 + parts[2];
      return parts[0];
    }

      if (idx !== -1) allTasks[idx] = res.data; else allTasks.push(res.data);
      closeModal('taskModal');
      refreshAllViews();
      showToast('保存しました');
    }

    async function deleteTaskItem(task, btn) {
      const childrenCount = allTasks.filter(function(c) { return c.parent_id === task.id; }).length;
      const extra = childrenCount > 0 ? '\n※中に含まれる' + childrenCount + '件の項目も一緒に削除されます。' : '';
      if (!confirm('「' + task.name + '」を削除しますか？' + extra)) return;

      btn.disabled = true;
      const r1 = await db.from('tasks').delete().eq('parent_id', task.id);
      const r2 = await db.from('tasks').delete().eq('id', task.id);
      if (r1.error || r2.error) {
        alert('削除失敗：' + ((r1.error || r2.error).message));
        btn.disabled = false;
        return;
      }

      allTasks = allTasks.filter(function(t) { return t.id !== task.id && t.parent_id !== task.id; });
      refreshAllViews();
      showToast('削除しました');
    }

    // ==================== CRUD：メモ ====================
    function openMemoModal(memo) {
      const proj = getCurrentProject();
      if (!proj) { alert('先に作品を作成してください'); return; }
      editingMemoId = memo ? memo.id : null;
      document.getElementById('memoModalTitle').textContent = memo ? 'メモを編集' : '編集・修正メモを書く';

      const taskSel = document.getElementById('memoTaskSelect');
      taskSel.innerHTML = '<option value="">作品全体へのメモ（指定なし）</option>';
      allTasks.filter(function(t) { return t.project_id === proj.id && !t.parent_id; }).forEach(function(p) {
        const pOpt = document.createElement('option');
        pOpt.value = p.id;
        pOpt.textContent = '【' + (proj.level2_label || 'シーン') + '】' + formatTaskDisplayName(p);
        if (memo && memo.task_id === p.id) pOpt.selected = true;
        taskSel.appendChild(pOpt);

        allTasks.filter(function(c) { return c.parent_id === p.id; }).forEach(function(c) {
          const cOpt = document.createElement('option');
          cOpt.value = c.id;
          cOpt.textContent = ' └ ' + formatTaskDisplayName(c);
          if (memo && memo.task_id === c.id) cOpt.selected = true;
          taskSel.appendChild(cOpt);
        });
      });

      document.getElementById('memoTcInput').value = memo ? (memo.timecode || '') : '';
      document.getElementById('memoCatInput').value = memo ? memo.category : '#カット';
      document.getElementById('memoContentInput').value = memo ? memo.content : '';
      populateAssigneeSelect(document.getElementById('memoAssigneeSelect'), memo ? memo.assignee : '', '全員（指定なし）');
      document.getElementById('memoStatusInput').value = memo ? memo.status : '未対応';
      document.getElementById('memoModalMsg').textContent = '';
      document.getElementById('memoModal').classList.add('open');
    }
